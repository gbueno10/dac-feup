# Gridfinity Generator Web App

Ferramenta web para gerar arquivos Gridfinity baseados nas medidas reais de uma gaveta, otimizado para Bambu Lab A1 Mini.

## Funcionalidades

- **Smart Tiling**: Divide automaticamente baseplates grandes em peças que cabem na A1 Mini (max 4x4 unidades)
- **Geração de Bins**: Crie caixas Gridfinity personalizadas
- **Validação Automática**: Alertas de incompatibilidade para peças muito grandes
- **Download Unificado**: Arquivo ZIP com todas as peças

## Instalação

```bash
# Criar ambiente virtual
python -m venv venv
source venv/bin/activate  # Linux/Mac
# ou
venv\Scripts\activate  # Windows

# Instalar dependências
pip install -r requirements.txt
```

## Execução

```bash
streamlit run app.py
```

## Hardware Alvo

- **Bambu Lab A1 Mini**: Volume 180x180x180mm
- Limite seguro: 4x4 unidades (168mm)

## Especificações Gridfinity

- Unidade base: 42mm x 42mm
- Altura da unidade: 7mm

"""
Gera peças de teste para validar o encaixe Dovetail (Rabo de Andorinha).
Cria dois mini baseplates 1x1 com conectores macho e fêmea.
"""

from build123d import *

# Constantes do Gridfinity
GRID_UNIT = 42.0
BASEPLATE_HEIGHT = 4.65
BASEPLATE_CORNER_RADIUS = 3.75

# Constantes do Dovetail
DOVETAIL_DEPTH = 2.0
DOVETAIL_WIDTH_MAX = 6.0
DOVETAIL_WIDTH_MIN = 4.0
DOVETAIL_TOLERANCE = 0.20


def create_dovetail_solid(height: float, tolerance: float = 0.0):
    """Cria o sólido do conector dovetail."""
    depth = DOVETAIL_DEPTH
    width_max = DOVETAIL_WIDTH_MAX
    width_min = DOVETAIL_WIDTH_MIN
    
    w_max = width_max + tolerance
    w_min = width_min + tolerance
    d = depth + tolerance
    
    connector_height = height / 2  # Meia altura
    start_x = -0.5 if tolerance > 0 else 0
    
    with BuildPart() as dovetail:
        with BuildSketch(Plane.XY):
            with BuildLine():
                pts = [
                    (start_x, -w_min/2),
                    (start_x, w_min/2),
                    (start_x + d, w_max/2),
                    (start_x + d, -w_max/2),
                    (start_x, -w_min/2)
                ]
                Polyline(pts)
            make_face()
        extrude(amount=connector_height)
    
    return dovetail.part


def create_test_piece_male():
    """Cria peça de teste 1x1 com conector MACHO na direita."""
    grid_size = GRID_UNIT
    
    with BuildPart() as piece:
        # Base 1x1
        with BuildSketch(Plane.XY):
            RectangleRounded(grid_size, grid_size, BASEPLATE_CORNER_RADIUS)
        extrude(amount=BASEPLATE_HEIGHT)
        
        # Conector macho na borda direita

"""
Gerador de Baseplates Gridfinity com Travamento Dovetail (Rabo de Andorinha)
Otimizado para Bambu Lab A1 Mini - Versão Standalone
"""

from build123d import *
from typing import List, Tuple, Optional, Dict
from .constants import *

class BasePlateGenerator:
    def __init__(self, optimize_for_a1_mini: bool = True):
        self.optimize_for_a1_mini = optimize_for_a1_mini
        self.max_units = A1_MINI_MAX_UNITS if optimize_for_a1_mini else 20

    # --- Lógica de Cálculo ---

    def calculate_grid_from_drawer(self, drawer_x_mm: float, drawer_y_mm: float) -> Tuple[int, int, float, float]:
        """Calcula unidades e a sobra total em cada eixo."""
        units_x = max(1, int(drawer_x_mm // GRID_UNIT))
        units_y = max(1, int(drawer_y_mm // GRID_UNIT))
        
        # Sobra para espaçadores
        sobra_x = drawer_x_mm - (units_x * GRID_UNIT)
        sobra_y = drawer_y_mm - (units_y * GRID_UNIT)
        
        # Margem de segurança
        if sobra_x > 1.0: sobra_x -= 0.5 
        else: sobra_x = 0
        if sobra_y > 1.0: sobra_y -= 0.5
        else: sobra_y = 0

        return units_x, units_y, sobra_x, sobra_y

    def calculate_tiling_strategy(self, total_x: int, total_y: int, spacer_x: float = 0, spacer_y: float = 0) -> List[Tuple[int, int, int, int]]:
        """
        Calcula a estratégia de divisão em tiles.
        
        IMPORTANTE: Considera spacers e conectores para garantir que cada peça caiba na mesa.
        Fórmula do limite: grid + spacer + conector_macho ≤ 180mm
        """
        # Calcula max_units efetivo considerando spacers e conectores
        # Peças de CANTO têm spacer de um lado + conector do outro
        # Pior caso: spacer + grid + conector ≤ 180
        max_spacer = max(spacer_x, spacer_y)
        available_for_grid = A1_MINI_BUILD_VOLUME_X - max_spacer - self.DOVETAIL_DEPTH
        effective_max_units = int(available_for_grid // GRID_UNIT)
        effective_max_units = max(1, min(effective_max_units, self.max_units))
        
        tiles = []
        for x in range(0, total_x, effective_max_units):
            for y in range(0, total_y, effective_max_units):
                chunk_x = min(effective_max_units, total_x - x)
                chunk_y = min(effective_max_units, total_y - y)
                tiles.append((x, y, chunk_x, chunk_y))
        return tiles

    # --- Geometria do Dovetail (Rabo de Andorinha) ---
    
    # Constantes do conector (acessíveis para cálculos)
    DOVETAIL_DEPTH = 2.0  # Profundidade do conector macho (quanto sai da peça)

    def _create_dovetail_solid(self, height: float, tolerance: float = 0.0) -> Part:
        """
        Cria o sólido 3D do rabo de andorinha (trapézio pequeno e discreto).
        O conector é criado apontando para +X, com base em X=0.
        
        IMPORTANTE: O conector ocupa apenas METADE da altura (parte inferior).
        Isso mantém a estrutura mais sólida e facilita a impressão.
        
        Args:
            height: Altura TOTAL da peça (o conector usará metade)
            tolerance: Folga adicional para o corte fêmea (0.0 para macho, 0.20 para fêmea)
        
        Dimensões otimizadas para FDM:
        - Profundidade: 2.0mm (avanço)
        - Largura na ponta: 6.0mm
        - Largura no pescoço: 4.0mm
        - Altura: metade da altura da peça
        """
        # Dimensões do Conector
        depth = 2.0       # Profundidade (avanço)
        width_max = 6.0   # Largura na ponta (maior)
        width_min = 4.0   # Largura no pescoço (menor)
        
        # Aplica tolerância (aumenta o buraco fêmea)
        w_max = width_max + tolerance
        w_min = width_min + tolerance
        d = depth + tolerance  # Fêmea um pouco mais funda também
        
        # MEIA ALTURA - conector só na parte inferior
        connector_height = height / 2
        
        # Desenha o trapézio no plano XY, apontando para +X
        # Para o FÊMEA funcionar, precisamos que a base comece um pouco ANTES da borda
        start_x = -0.5 if tolerance > 0 else 0  # Fêmea começa 0.5mm antes
        
        with BuildPart() as dovetail:
            with BuildSketch(Plane.XY):
                with BuildLine():
                    pts = [
                        (start_x, -w_min/2),           # Base esquerda baixo
                        (start_x, w_min/2),            # Base esquerda cima
                        (start_x + d, w_max/2),        # Ponta cima
                        (start_x + d, -w_max/2),       # Ponta baixo
                        (start_x, -w_min/2)            # Fecha
                    ]
                    Polyline(pts)
                make_face()
            # Extrude apenas MEIA ALTURA (parte inferior, Z=0 até Z=height/2)
            extrude(amount=connector_height)
        
        return dovetail.part

    # --- Geometria do Baseplate ---

    def _create_simple_baseplate(self, size_x: int, size_y: int, tiling_info: Dict, spacers: Dict) -> Part:
        """
        Gera o baseplate com conectores Dovetail nas bordas apropriadas.
        
        IMPORTANTE: Os conectores são gerados POR CÉLULA, não por placa!
        Uma placa 4x4 terá 4 conectores em cada borda de interface.
        
        tiling_info: Diz se tem vizinho na {has_left, has_right, has_top, has_bottom}
        """
        grid_width = size_x * GRID_UNIT
        grid_depth = size_y * GRID_UNIT
        
        # Dimensões totais com espaçadores
        total_width = grid_width + spacers['left'] + spacers['right']
        total_depth = grid_depth + spacers['top'] + spacers['bottom']
        
        # Ajuste do centro geométrico
        center_offset_x = (spacers['right'] - spacers['left']) / 2
        center_offset_y = (spacers['top'] - spacers['bottom']) / 2

        # Altura do conector (mesma altura da base)
        conn_height = BASEPLATE_HEIGHT

        with BuildPart() as base:
            # 1. Base Sólida Principal
            with BuildSketch(Plane.XY):
                with Locations((center_offset_x, center_offset_y)):
                    RectangleRounded(total_width, total_depth, BASEPLATE_CORNER_RADIUS)
            extrude(amount=BASEPLATE_HEIGHT)

            # 2. Recortes Gridfinity (Padrão) + Conectores Dovetail POR CÉLULA
            for ix in range(size_x):
                for iy in range(size_y):
                    # Centro desta célula (relativo ao centro do grid)
                    cx = (ix - (size_x - 1) / 2) * GRID_UNIT
                    cy = (iy - (size_y - 1) / 2) * GRID_UNIT
                    
                    # === Recortes Gridfinity (padrão) ===
                    # Top Lip
                    with BuildSketch(Plane.XY.offset(BASEPLATE_HEIGHT)):
                        with Locations((cx, cy)):
                            RectangleRounded(GRID_UNIT - 0.5, GRID_UNIT - 0.5, BASEPLATE_CORNER_RADIUS - 0.25)
                    extrude(amount=-BASEPLATE_LIP_HEIGHT, mode=Mode.SUBTRACT)

                    # Bottom Square
                    with BuildSketch(Plane.XY.offset(BASEPLATE_HEIGHT - BASEPLATE_LIP_HEIGHT)):
                        with Locations((cx, cy)):
                            RectangleRounded(
                                GRID_UNIT - 0.5 - BASEPLATE_LIP_WIDTH*2, 
                                GRID_UNIT - 0.5 - BASEPLATE_LIP_WIDTH*2, 
                                BASEPLATE_CORNER_RADIUS - 0.5
                            )
                    extrude(amount=-(BASEPLATE_HEIGHT - BASEPLATE_LIP_HEIGHT), mode=Mode.SUBTRACT)

                    # === Conectores Dovetail POR CÉLULA ===
                    # Regra de Polaridade:
                    # - Direita (+X) e Cima (+Y): MACHO (Union)
                    # - Esquerda (-X) e Baixo (-Y): FÊMEA (Subtract com tolerância)
                    
                    # --- BORDA DIREITA: Se esta célula está na última coluna E tem vizinho à direita ---
                    if ix == (size_x - 1) and tiling_info.get('has_right'):
                        male = self._create_dovetail_solid(conn_height, tolerance=0.0)
                        # Posiciona na borda direita do grid, alinhado ao centro Y desta célula
                        male = male.move(Location((grid_width/2, cy, 0)))
                        add(male, mode=Mode.ADD)
                    
                    # --- BORDA ESQUERDA: Se esta célula está na primeira coluna E tem vizinho à esquerda ---
                    if ix == 0 and tiling_info.get('has_left'):
                        female = self._create_dovetail_solid(conn_height, tolerance=0.20)
                        # O sólido aponta para +X. Para cortar na esquerda, precisamos que ele
                        # ENTRE na peça vindo de fora. Posicionamos a BASE do trapézio na borda
                        # e ele avança para DENTRO (+X).
                        # Posição X: -grid_width/2 (borda esquerda, base do trapézio começa aqui)
                        female = female.move(Location((-grid_width/2, cy, 0)))
                        add(female, mode=Mode.SUBTRACT)
                    
                    # --- BORDA SUPERIOR: Se esta célula está na última linha E tem vizinho acima ---
                    if iy == (size_y - 1) and tiling_info.get('has_top'):
                        male = self._create_dovetail_solid(conn_height, tolerance=0.0)
                        # Rotaciona 90° para apontar para +Y
                        male = male.rotate(Axis.Z, 90)
                        # Posiciona na borda superior, alinhado ao centro X desta célula
                        male = male.move(Location((cx, grid_depth/2, 0)))
                        add(male, mode=Mode.ADD)
                    
                    # --- BORDA INFERIOR: Se esta célula está na primeira linha E tem vizinho abaixo ---
                    if iy == 0 and tiling_info.get('has_bottom'):
                        female = self._create_dovetail_solid(conn_height, tolerance=0.20)
                        # Rotaciona 90° para apontar para +Y (entrando na peça de baixo para cima)
                        female = female.rotate(Axis.Z, 90)
                        # Posiciona na borda inferior, alinhado ao centro X desta célula
                        female = female.move(Location((cx, -grid_depth/2, 0)))
                        add(female, mode=Mode.SUBTRACT)

        return base.part

    def _center_part_on_origin(self, part: Part) -> Part:
        """
        Centraliza a peça na origem (0,0) para facilitar alinhamento no slicer.
        O Z fica no chão (z=0).
        """
        bb = part.bounding_box()
        center_x = (bb.min.X + bb.max.X) / 2
        center_y = (bb.min.Y + bb.max.Y) / 2
        min_z = bb.min.Z
        
        return part.move(Location((-center_x, -center_y, -min_z)))

    # --- Fluxo Principal ---

    def generate_from_drawer(
        self, 
        drawer_x_mm: float, 
        drawer_y_mm: float, 
        include_connectors: bool = True,  # Mantido para compatibilidade (agora intrínseco)
        center_on_origin: bool = True
    ) -> Tuple[Compound, int, int, List[Tuple[int, int, int, int]], int]:
        
        # 1. Calcula Grid e Sobras
        units_x, units_y, sobra_x, sobra_y = self.calculate_grid_from_drawer(drawer_x_mm, drawer_y_mm)
        spacer_x = sobra_x / 2
        spacer_y = sobra_y / 2
        
        # 2. Define Tiles (passa spacers para cálculo correto do limite)
        tiles = self.calculate_tiling_strategy(units_x, units_y, spacer_x, spacer_y)
        
        parts_export = []
        spacing = A1_MINI_MAX_UNITS * GRID_UNIT + 20

        # 3. Gera cada Tile
        for idx, (pos_x, pos_y, size_x, size_y) in enumerate(tiles):
            # Informação de Vizinhança (Tiling Info)
            tiling_info = {
                'has_right': (pos_x + size_x) < units_x,
                'has_left': pos_x > 0,
                'has_top': (pos_y + size_y) < units_y,
                'has_bottom': pos_y > 0
            }

            # Spacers (Abas) - Só aplicados nas bordas externas que tocam a gaveta
            my_spacers = {
                'left': spacer_x if not tiling_info['has_left'] else 0,
                'right': spacer_x if not tiling_info['has_right'] else 0,
                'bottom': spacer_y if not tiling_info['has_bottom'] else 0,
                'top': spacer_y if not tiling_info['has_top'] else 0
            }

            # Gera a peça
            part = self._create_simple_baseplate(size_x, size_y, tiling_info, my_spacers)
            
            # Centraliza na origem se solicitado
            if center_on_origin:
                part = self._center_part_on_origin(part)
            
            # Posiciona para exportação (visualização explodida no slicer)
            if len(tiles) > 1:
                loc_x = (idx % 3) * spacing
                loc_y = (idx // 3) * spacing
                part_copy = Part(part.wrapped).move(Location((loc_x, loc_y, 0)))
            else:
                part_copy = Part(part.wrapped)
            
            parts_export.append(part_copy)

        # Retorna Compound (0 conectores avulsos, pois agora são integrados)
        return Compound(children=parts_export), units_x, units_y, tiles, 0

    def get_generation_summary(self, ux, uy, tiles, n_conn):
        return f"Grid {ux}x{uy} ({len(tiles)} partes com encaixe Dovetail)"

    def get_tiling_description(self, tiles):
        if len(tiles) == 1: 
            return "Peça única."
        return f"Dividido em {len(tiles)} partes intertravadas."

    def needs_tiling(self, ux, uy):
        return ux > self.max_units or uy > self.max_units

"""
Gerador de Bins (Caixas) Gridfinity
"""

from build123d import *
from typing import Tuple, Optional
from .constants import *


class BinGenerator:
    """
    Gera bins (caixas) Gridfinity personalizados.
    """
    
    def __init__(self, optimize_for_a1_mini: bool = True):
        """
        Inicializa o gerador de bins.
        
        Args:
            optimize_for_a1_mini: Se True, valida contra limite de 4x4 unidades
        """
        self.optimize_for_a1_mini = optimize_for_a1_mini
        self.max_units = A1_MINI_MAX_UNITS if optimize_for_a1_mini else 999
    
    def validate_size(self, units_x: int, units_y: int) -> Tuple[bool, str]:
        """
        Valida se o tamanho do bin é compatível com a impressora.
        
        Args:
            units_x: Unidades no eixo X
            units_y: Unidades no eixo Y
            
        Returns:
            Tuple com (is_valid, message)
        """
        if not self.optimize_for_a1_mini:
            return True, "Validação desativada."
        
        width_mm = units_x * GRID_UNIT
        depth_mm = units_y * GRID_UNIT
        
        if units_x > self.max_units or units_y > self.max_units:
            return False, (
                f"⚠️ ALERTA DE INCOMPATIBILIDADE: "
                f"Bin {units_x}x{units_y} ({width_mm:.0f}x{depth_mm:.0f}mm) "
                f"excede o limite da A1 Mini (máximo {self.max_units}x{self.max_units} = "
                f"{self.max_units * GRID_UNIT:.0f}mm). "
                f"Diferente de baseplates, bins não podem ser divididos e remontados."
            )
        
        return True, f"✅ Bin {units_x}x{units_y} compatível com A1 Mini."
    
    def generate_bin(
        self,
        units_x: int,
        units_y: int,
        height_units: int = 3,
        with_label: bool = False,
        wall_thickness: float = BIN_WALL_THICKNESS
    ) -> Part:
        """
        Gera um bin Gridfinity com perfil de encaixe correto (por célula).
        
        Args:
            units_x: Número de unidades no eixo X
            units_y: Número de unidades no eixo Y
            height_units: Altura em unidades (1 unidade = 7mm)
            with_label: Adiciona área para etiqueta na frente
            wall_thickness: Espessura da parede
            
        Returns:
            Objeto Part representando o bin
        """
        width = units_x * GRID_UNIT - 0.5  # Folga para encaixe
        depth = units_y * GRID_UNIT - 0.5
        height = height_units * GRID_HEIGHT_UNIT
        
        internal_width = width - wall_thickness * 2
        internal_depth = depth - wall_thickness * 2
        
        # Constantes do perfil de encaixe (lip) - baseado no padrão Gridfinity
        # O perfil encaixa nos recortes do baseplate
        lip_base_size = GRID_UNIT - 0.5  # 41.5mm - tamanho do recorte no baseplate
        lip_inner_size = lip_base_size - BASEPLATE_LIP_WIDTH * 2  # Parte inferior do lip
        lip_total_height = BASEPLATE_HEIGHT  # 4.65mm - altura total do lip
        lip_step_height = BASEPLATE_LIP_HEIGHT  # 2.15mm - altura do degrau superior
        
        # Altura interna: do fundo (acima do lip + bottom) até o topo
        internal_height = height - lip_total_height - BIN_BOTTOM_THICKNESS
        
        with BuildPart() as bin_part:
            # 1. Corpo externo do bin (começa ACIMA do lip)
            with BuildSketch(Plane.XY.offset(lip_total_height)) as outer:
                RectangleRounded(
                    width=width,
                    height=depth,
                    radius=BIN_CORNER_RADIUS
                )
            extrude(amount=height - lip_total_height)
            
            # 2. Cavidade interna
            with BuildSketch(Plane.XY.offset(lip_total_height + BIN_BOTTOM_THICKNESS)) as inner:
                RectangleRounded(
                    width=internal_width,
                    height=internal_depth,
                    radius=BIN_CORNER_RADIUS - wall_thickness
                )
            extrude(amount=internal_height, mode=Mode.SUBTRACT)
            
            # 3. Lip (borda superior) para empilhamento
            lip_outer = BIN_LIP_WIDTH  # 2.6mm
            lip_height = BIN_LIP_HEIGHT  # 4.4mm
            
            with BuildSketch(Plane.XY.offset(height)) as lip:
                RectangleRounded(
                    width=width,
                    height=depth,
                    radius=BIN_CORNER_RADIUS
                )
                RectangleRounded(
                    width=width - lip_outer * 2,
                    height=depth - lip_outer * 2,
                    radius=BIN_CORNER_RADIUS - lip_outer,
                    mode=Mode.SUBTRACT
                )
            extrude(amount=-lip_height, mode=Mode.SUBTRACT)
            
            # 4. PERFIS DE ENCAIXE NO BASEPLATE (por célula!)
            # Cada célula do bin precisa de seu próprio perfil que encaixa no baseplate
            for ix in range(units_x):
                for iy in range(units_y):
                    # Centro desta célula (relativo ao centro do bin)
                    cx = (ix - (units_x - 1) / 2) * GRID_UNIT
                    cy = (iy - (units_y - 1) / 2) * GRID_UNIT
                    
                    # Parte SUPERIOR do lip (degrau) - encaixa no recorte superior do baseplate
                    # Tamanho: 41.5mm, altura: do topo (4.65mm) descendo 2.15mm
                    with BuildSketch(Plane.XY.offset(lip_total_height)):
                        with Locations((cx, cy)):
                            RectangleRounded(
                                lip_base_size,
                                lip_base_size,
                                BASEPLATE_CORNER_RADIUS - 0.25
                            )
                    extrude(amount=-lip_step_height)
                    
                    # Parte INFERIOR do lip (base mais estreita) - encaixa no recorte inferior
                    # Tamanho: 41.5 - 1.6 = 39.9mm, altura: 4.65 - 2.15 = 2.5mm
                    with BuildSketch(Plane.XY.offset(lip_total_height - lip_step_height)):
                        with Locations((cx, cy)):
                            RectangleRounded(
                                lip_inner_size,
                                lip_inner_size,
                                BASEPLATE_CORNER_RADIUS - 0.5
                            )
                    extrude(amount=-(lip_total_height - lip_step_height))
            
            # 5. Recursos opcionais - Label (etiqueta)
            if with_label:
                label_width = width * 0.8
                label_height = 12  # mm
                label_depth = 0.6  # profundidade do rebaixo
                
                with BuildSketch(Plane.XZ.offset(-depth / 2)) as label:
                    with Locations([(0, height - label_height / 2 - 2)]):
                        RectangleRounded(
                            width=label_width,
                            height=label_height,
                            radius=1.5
                        )
                extrude(amount=label_depth, mode=Mode.SUBTRACT)
        
        return bin_part.part
    
    def generate_divider_bin(
        self,
        units_x: int,
        units_y: int,
        height_units: int = 3,
        divisions_x: int = 1,
        divisions_y: int = 1
    ) -> Part:
        """
        Gera um bin com divisórias internas.
        
        Args:
            units_x: Unidades no eixo X
            units_y: Unidades no eixo Y
            height_units: Altura em unidades
            divisions_x: Número de divisões no eixo X
            divisions_y: Número de divisões no eixo Y
            
        Returns:
            Objeto Part representando o bin com divisórias
        """
        # Primeiro, gera o bin base
        base_bin = self.generate_bin(units_x, units_y, height_units)
        
        width = units_x * GRID_UNIT - 0.5
        depth = units_y * GRID_UNIT - 0.5
        height = height_units * GRID_HEIGHT_UNIT
        
        internal_width = width - BIN_WALL_THICKNESS * 2
        internal_depth = depth - BIN_WALL_THICKNESS * 2
        
        # Altura das divisórias: mesma altura da cavidade interna
        # Fundo = BASEPLATE_HEIGHT + BIN_BOTTOM_THICKNESS = 5.45mm
        # Topo = height (mesmo da cavidade, que vai até o topo do bin)
        divider_base_z = BASEPLATE_HEIGHT + BIN_BOTTOM_THICKNESS
        divider_height = height - divider_base_z  # Igual à cavidade interna
        
        divider_thickness = 1.0  # mm
        
        with BuildPart() as bin_with_dividers:
            add(base_bin)
            
            # Divisórias no eixo X
            # Usamos Plane.YZ no offset X, o retângulo tem (largura_Y, altura_Z)
            if divisions_x > 1:
                cell_width = internal_width / divisions_x
                for i in range(1, divisions_x):
                    x_pos = -internal_width / 2 + cell_width * i
                    # Desenha na base Z e extrude para cima
                    with BuildSketch(Plane.YZ.offset(x_pos)):
                        # Centro do retângulo no meio da altura da divisória
                        center_z = divider_base_z + divider_height / 2
                        with Locations((0, center_z)):
                            Rectangle(internal_depth, divider_height)
                    extrude(amount=divider_thickness / 2, both=True)
            
            # Divisórias no eixo Y
            if divisions_y > 1:
                cell_depth = internal_depth / divisions_y
                for i in range(1, divisions_y):
                    y_pos = -internal_depth / 2 + cell_depth * i
                    with BuildSketch(Plane.XZ.offset(y_pos)):
                        center_z = divider_base_z + divider_height / 2
                        with Locations((0, center_z)):
                            Rectangle(internal_width, divider_height)
                    extrude(amount=divider_thickness / 2, both=True)
        
        return bin_with_dividers.part

"""
Utilitários para exportação de arquivos STL e ZIP
"""

import io
import zipfile
import tempfile
import os
from typing import List, Tuple, Optional
from build123d import Compound, Part, export_stl


def export_to_stl_bytes(shape: Compound | Part, tolerance: float = 0.001) -> bytes:
    """
    Exporta um shape para bytes STL.
    
    Args:
        shape: Objeto Compound ou Part do build123d
        tolerance: Tolerância para tessellation
        
    Returns:
        Bytes do arquivo STL
    """
    with tempfile.NamedTemporaryFile(suffix='.stl', delete=False) as tmp:
        tmp_path = tmp.name
    
    try:
        export_stl(shape, tmp_path, tolerance=tolerance)
        with open(tmp_path, 'rb') as f:
            return f.read()
    finally:
        if os.path.exists(tmp_path):
            os.remove(tmp_path)


def create_download_zip(
    baseplate: Optional[Compound | Part] = None,
    bins: Optional[List[Tuple[str, Part]]] = None,
    baseplate_filename: str = "baseplate_full_set_a1mini.stl"
) -> bytes:
    """
    Cria arquivo ZIP contendo baseplate e bins.
    
    Args:
        baseplate: Compound com todas as peças do baseplate
        bins: Lista de tuplas (nome_arquivo, Part) para os bins
        baseplate_filename: Nome do arquivo do baseplate no ZIP
        
    Returns:
        Bytes do arquivo ZIP
    """
    zip_buffer = io.BytesIO()
    
    with zipfile.ZipFile(zip_buffer, 'w', zipfile.ZIP_DEFLATED) as zf:
        # Adiciona baseplate
        if baseplate is not None:
            stl_bytes = export_to_stl_bytes(baseplate)
            zf.writestr(baseplate_filename, stl_bytes)
        
        # Adiciona bins
        if bins:
            for filename, bin_part in bins:
                stl_bytes = export_to_stl_bytes(bin_part)
                zf.writestr(filename, stl_bytes)
    
    zip_buffer.seek(0)
    return zip_buffer.getvalue()


def get_file_size_str(size_bytes: int) -> str:
    """Converte bytes para string legível."""
    if size_bytes < 1024:
        return f"{size_bytes} B"
    elif size_bytes < 1024 * 1024:
        return f"{size_bytes / 1024:.1f} KB"
    else:
        return f"{size_bytes / (1024 * 1024):.1f} MB"


def export_parts_individually(
    compound: Compound,
    base_filename: str = "baseplate"
) -> List[Tuple[str, bytes]]:
    """
    Exporta cada peça de um Compound como STL individual.
    Cada peça é centrada na origem (0,0,0) para facilitar alinhamento no slicer.
    
    Args:
        compound: Compound contendo múltiplas Parts
        base_filename: Nome base para os arquivos (ex: "baseplate" -> "baseplate_1.stl")
        
    Returns:
        Lista de tuplas (nome_arquivo, bytes_stl)
    """
    from build123d import Location
    
    parts = []
    children = list(compound.solids())
    
    for i, solid in enumerate(children, 1):
        # Centraliza cada peça na origem
        bb = solid.bounding_box()
        center_x = (bb.min.X + bb.max.X) / 2
        center_y = (bb.min.Y + bb.max.Y) / 2
        min_z = bb.min.Z
        
        # Move para centro em XY e Z=0
        centered = solid.move(Location((-center_x, -center_y, -min_z)))
        
        # Exporta
        stl_bytes = export_to_stl_bytes(centered)
        filename = f"{base_filename}_{i}.stl"
        parts.append((filename, stl_bytes))
    
    return parts


def create_individual_parts_zip(
    compound: Compound,
    base_filename: str = "baseplate"
) -> bytes:
    """
    Cria ZIP com cada peça exportada individualmente e centrada.
    Ideal para importar no slicer com alinhamento automático.
    
    Args:
        compound: Compound com múltiplas peças
        base_filename: Nome base para os arquivos
        
    Returns:
        Bytes do arquivo ZIP
    """
    zip_buffer = io.BytesIO()
    
    parts = export_parts_individually(compound, base_filename)
    
    with zipfile.ZipFile(zip_buffer, 'w', zipfile.ZIP_DEFLATED) as zf:
        for filename, stl_bytes in parts:
            zf.writestr(filename, stl_bytes)
    
    zip_buffer.seek(0)
    return zip_buffer.getvalue()
