# Gridfinity Generator Web App

Ferramenta web para gerar arquivos Gridfinity baseados nas medidas reais de uma gaveta, otimizado para Bambu Lab A1 Mini.

## Funcionalidades

- **Smart Tiling**: Divide automaticamente baseplates grandes em peças que cabem na A1 Mini (max 4x4 unidades)
- **Geração de Bins**: Crie caixas Gridfinity personalizadas
- **Validação Automática**: Alertas de incompatibilidade para peças muito grandes
- **Download Unificado**: Arquivo ZIP com todas as peças

## Instalação

```bash
# Criar ambiente virtual
python -m venv venv
source venv/bin/activate  # Linux/Mac
# ou
venv\Scripts\activate  # Windows

# Instalar dependências
pip install -r requirements.txt
```

## Execução

```bash
streamlit run app.py
```

## Hardware Alvo

- **Bambu Lab A1 Mini**: Volume 180x180x180mm
- Limite seguro: 4x4 unidades (168mm)

## Especificações Gridfinity

- Unidade base: 42mm x 42mm
- Altura da unidade: 7mm

"""
Gerador de Baseplates Gridfinity - Versão Standalone (Sem dependências extras)
Desenha a geometria "na mão" para garantir compatibilidade e controle total.
"""

from build123d import *
from typing import List, Tuple, Optional
from .constants import *

class BasePlateGenerator:
    def __init__(self, optimize_for_a1_mini: bool = True):
        self.optimize_for_a1_mini = optimize_for_a1_mini
        self.max_units = A1_MINI_MAX_UNITS if optimize_for_a1_mini else 20

    # --- Lógica de Cálculo (Matemática) ---

    def calculate_grid_from_drawer(self, drawer_x_mm: float, drawer_y_mm: float) -> Tuple[int, int]:
        units_x = int(drawer_x_mm // GRID_UNIT)
        units_y = int(drawer_y_mm // GRID_UNIT)
        return max(1, units_x), max(1, units_y)

    def calculate_tiling_strategy(self, total_x: int, total_y: int) -> List[Tuple[int, int, int, int]]:
        tiles = []
        for x in range(0, total_x, self.max_units):
            for y in range(0, total_y, self.max_units):
                chunk_x = min(self.max_units, total_x - x)
                chunk_y = min(self.max_units, total_y - y)
                tiles.append((x, y, chunk_x, chunk_y))
        return tiles

    # --- Geometria dos Conectores (Gridfinity Refined) ---

    def _get_connector_cutout_shape(self) -> Part:
        """Cria o formato do BURACO do conector (fêmea)"""
        middle_width = 6.0 / 2
        middle_height = 3.0
        thickness = 10.0 # Profundidade exagerada para garantir o corte
        
        with BuildPart() as part:
            with BuildSketch() as sketch:
                with BuildLine():
                    l0 = Line((0, 0), (middle_width, 0))
                    l1 = Line(l0 @ 1, (middle_width, middle_height))
                    l2 = Line(l1 @ 1, ((l1 @ 1).X + 4, (l1 @ 1).Y + 3))
                    l3 = Line(l2 @ 1, ((l2 @ 1).X, (l2 @ 1).Y + 3))
                    Line(l3 @ 1, (0, (l3 @ 1).Y))
                    mirror(about=Plane.YZ)
                make_face()
            extrude(sketch.sketch, thickness)
            
        # Rotaciona para ficar na orientação certa para cortar a lateral
        return part.part

    def _create_connector_clip(self) -> Part:
        """Cria o CLIP físico (macho) para impressão"""
        middle_width = 5.8 / 2
        middle_height = 6.5 / 2
        thickness = 2.8
        chamfer_value = 0.4

        with BuildPart() as part:
            with BuildSketch() as sketch:
                with BuildLine():
                    l1 = Line((middle_width, 0), (middle_width, middle_height))
                    l2 = Line(l1 @ 1, ((l1 @ 1).X + 3.9, (l1 @ 1).Y + 2.92))
                    l3 = Line(l2 @ 1, ((l2 @ 1).X, (l2 @ 1).Y + 2.53))
                    Line(l3 @ 1, (0, (l3 @ 1).Y))
                    mirror(about=Plane.XZ)
                    mirror(about=Plane.YZ)
                make_face()
            extrude(sketch.sketch, thickness)
            try:
                faces = [face.edges() for face in part.faces().filter_by(Axis.Z)]
                chamfer(faces, chamfer_value)
            except:
                pass 
        return part.part

    # --- Geometria do Baseplate (Modo Verboso/Manual) ---

    def _create_simple_baseplate(self, size_x: int, size_y: int, has_cutouts: dict) -> Part:
        """
        Cria o baseplate manualmente (RectangleRounded + Extrude).
        has_cutouts: dict indicando onde cortar {'top': bool, 'bottom': bool...}
        """
        width = size_x * GRID_UNIT
        depth = size_y * GRID_UNIT
        
        # 1. Base Sólida
        with BuildPart() as base:
            with BuildSketch():
                RectangleRounded(width, depth, BASEPLATE_CORNER_RADIUS)
            extrude(amount=BASEPLATE_HEIGHT)

            # 2. Recortes dos "Squares" (onde a caixa encaixa)
            # Desenhamos um perfil e subtraímos
            for ix in range(size_x):
                for iy in range(size_y):
                    # Calcula posição
                    cx = (ix - (size_x - 1) / 2) * GRID_UNIT
                    cy = (iy - (size_y - 1) / 2) * GRID_UNIT
                    
                    # Corte superior (o chanfro grande)
                    with BuildSketch(Plane.XY.offset(BASEPLATE_HEIGHT)):
                        with Locations((cx, cy)):
                            RectangleRounded(GRID_UNIT - 0.5, GRID_UNIT - 0.5, BASEPLATE_CORNER_RADIUS - 0.25)
                    extrude(amount=-BASEPLATE_LIP_HEIGHT, mode=Mode.SUBTRACT)

                    # Corte inferior (o quadrado interno)
                    with BuildSketch(Plane.XY.offset(BASEPLATE_HEIGHT - BASEPLATE_LIP_HEIGHT)):
                        with Locations((cx, cy)):
                            RectangleRounded(
                                GRID_UNIT - 0.5 - BASEPLATE_LIP_WIDTH*2, 
                                GRID_UNIT - 0.5 - BASEPLATE_LIP_WIDTH*2, 
                                BASEPLATE_CORNER_RADIUS - 0.5
                            )
                    extrude(amount=-(BASEPLATE_HEIGHT - BASEPLATE_LIP_HEIGHT), mode=Mode.SUBTRACT)

            # 3. Adiciona os cortes laterais para conectores (Se necessário)
            # A lógica aqui é posicionar o objeto de corte na borda correta e subtrair
            cutout_tool = self._get_connector_cutout_shape()
            
            # Se tiver corte na direita (+X)
            if has_cutouts.get('right'):
                # Rotaciona para apontar para X
                tool = cutout_tool.rotate(Axis.X, -90).rotate(Axis.Z, -90)
                # Posiciona na borda direita
                tool = tool.move(Location((width/2, 0, BASEPLATE_HEIGHT/2)))
                add(tool, mode=Mode.SUBTRACT)

            # Se tiver corte na esquerda (-X)
            if has_cutouts.get('left'):
                tool = cutout_tool.rotate(Axis.X, -90).rotate(Axis.Z, 90)
                tool = tool.move(Location((-width/2, 0, BASEPLATE_HEIGHT/2)))
                add(tool, mode=Mode.SUBTRACT)

            # Se tiver corte em cima (+Y)
            if has_cutouts.get('top'):
                tool = cutout_tool.rotate(Axis.X, -90).rotate(Axis.Z, 180)
                tool = tool.move(Location((0, depth/2, BASEPLATE_HEIGHT/2)))
                add(tool, mode=Mode.SUBTRACT)

            # Se tiver corte em baixo (-Y)
            if has_cutouts.get('bottom'):
                tool = cutout_tool.rotate(Axis.X, -90)
                tool = tool.move(Location((0, -depth/2, BASEPLATE_HEIGHT/2)))
                add(tool, mode=Mode.SUBTRACT)

        return base.part

    # --- Fluxo Principal ---

    def generate_from_drawer(
        self, 
        drawer_x_mm: float, 
        drawer_y_mm: float, 
        include_connectors: bool = True
    ) -> Tuple[Compound, int, int, List[Tuple[int, int, int, int]], int]:
        
        # 1. Calcula Grid
        units_x, units_y = self.calculate_grid_from_drawer(drawer_x_mm, drawer_y_mm)
        
        # 2. Define Tiles
        tiles = self.calculate_tiling_strategy(units_x, units_y)
        
        parts_export = []
        spacing = A1_MINI_MAX_UNITS * GRID_UNIT + 10

        # 3. Gera cada Tile
        for idx, (pos_x, pos_y, size_x, size_y) in enumerate(tiles):
            # Lógica para saber onde colocar furos de conector
            # "Tenho alguém à minha direita?" -> Se (minha posição X + meu tamanho) < total X
            has_right = (pos_x + size_x) < units_x
            has_left = pos_x > 0
            has_top = (pos_y + size_y) < units_y
            has_bottom = pos_y > 0
            
            cutouts = {
                'right': has_right,
                'left': has_left,
                'top': has_top,
                'bottom': has_bottom
            }

            # Gera a peça
            part = self._create_simple_baseplate(size_x, size_y, cutouts)
            
            # Posiciona para exportação (cria cópia independente)
            loc_x = (idx % 3) * spacing
            loc_y = (idx // 3) * spacing
            part_copy = Part(part.wrapped).move(Location((loc_x, loc_y, 0)))
            parts_export.append(part_copy)

        # 4. Gera Clipes (Macho)
        num_connectors = 0
        if len(tiles) > 1 and include_connectors:
            clip = self._create_connector_clip()
            # Estimativa de clipes necessários (um pouco a mais por segurança)
            num_connectors = len(tiles) * 2 
            
            for i in range(num_connectors):
                # Cria cópia independente do conector
                clip_copy = Part(clip.wrapped).move(Location((-50, i * 20, 0)))
                parts_export.append(clip_copy)

        return Compound(children=parts_export), units_x, units_y, tiles, num_connectors

    def get_tiling_description(self, tiles):
        if len(tiles) == 1:
            return "Peça única."
        return f"Dividido em {len(tiles)} partes."

    def get_generation_summary(self, ux, uy, tiles, n_conn):
        return f"Grid {ux}x{uy} ({len(tiles)} partes + {n_conn} conectores)"

    def needs_tiling(self, ux, uy):
        return ux > self.max_units or uy > self.max_units

"""
Gerador de Bins (Caixas) Gridfinity
"""

from build123d import *
from typing import Tuple, Optional
from .constants import *


class BinGenerator:
    """
    Gera bins (caixas) Gridfinity personalizados.
    """
    
    def __init__(self, optimize_for_a1_mini: bool = True):
        """
        Inicializa o gerador de bins.
        
        Args:
            optimize_for_a1_mini: Se True, valida contra limite de 4x4 unidades
        """
        self.optimize_for_a1_mini = optimize_for_a1_mini
        self.max_units = A1_MINI_MAX_UNITS if optimize_for_a1_mini else 999
    
    def validate_size(self, units_x: int, units_y: int) -> Tuple[bool, str]:
        """
        Valida se o tamanho do bin é compatível com a impressora.
        
        Args:
            units_x: Unidades no eixo X
            units_y: Unidades no eixo Y
            
        Returns:
            Tuple com (is_valid, message)
        """
        if not self.optimize_for_a1_mini:
            return True, "Validação desativada."
        
        width_mm = units_x * GRID_UNIT
        depth_mm = units_y * GRID_UNIT
        
        if units_x > self.max_units or units_y > self.max_units:
            return False, (
                f"⚠️ ALERTA DE INCOMPATIBILIDADE: "
                f"Bin {units_x}x{units_y} ({width_mm:.0f}x{depth_mm:.0f}mm) "
                f"excede o limite da A1 Mini (máximo {self.max_units}x{self.max_units} = "
                f"{self.max_units * GRID_UNIT:.0f}mm). "
                f"Diferente de baseplates, bins não podem ser divididos e remontados."
            )
        
        return True, f"✅ Bin {units_x}x{units_y} compatível com A1 Mini."
    
    def generate_bin(
        self,
        units_x: int,
        units_y: int,
        height_units: int = 3,
        with_label: bool = False,
        with_scoop: bool = False,
        wall_thickness: float = BIN_WALL_THICKNESS
    ) -> Part:
        """
        Gera um bin Gridfinity.
        
        Args:
            units_x: Número de unidades no eixo X
            units_y: Número de unidades no eixo Y
            height_units: Altura em unidades (1 unidade = 7mm)
            with_label: Adiciona área para etiqueta na frente
            with_scoop: Adiciona rampa para facilitar pegar itens
            wall_thickness: Espessura da parede
            
        Returns:
            Objeto Part representando o bin
        """
        width = units_x * GRID_UNIT - 0.5  # Folga para encaixe
        depth = units_y * GRID_UNIT - 0.5
        height = height_units * GRID_HEIGHT_UNIT
        
        with BuildPart() as bin_part:
            # Corpo externo do bin
            with BuildSketch():
                RectangleRounded(
                    width=width,
                    height=depth,
                    radius=BIN_CORNER_RADIUS
                )
            extrude(amount=height)
            
            # Cavidade interna
            internal_width = width - wall_thickness * 2
            internal_depth = depth - wall_thickness * 2
            internal_height = height - BIN_BOTTOM_THICKNESS
            
            with BuildSketch(Plane.XY.offset(BIN_BOTTOM_THICKNESS)):
                RectangleRounded(
                    width=internal_width,
                    height=internal_depth,
                    radius=BIN_CORNER_RADIUS - wall_thickness
                )
            extrude(amount=internal_height, mode=Mode.SUBTRACT)
            
            # Lip (borda) para empilhamento
            self._add_stacking_lip(bin_part, width, depth, height)
            
            # Base com perfil de encaixe no baseplate
            self._add_base_profile(bin_part, width, depth)
            
            # Recursos opcionais
            if with_scoop:
                self._add_scoop(bin_part, internal_width, internal_depth, internal_height)
            
            if with_label:
                self._add_label_area(bin_part, width, height)
        
        return bin_part.part
    
    def _add_stacking_lip(
        self,
        builder,
        width: float,
        depth: float,
        height: float
    ):
        """Adiciona lip para empilhamento."""
        lip_outer = 2.6
        lip_height = 4.4
        
        with BuildSketch(Plane.XY.offset(height)):
            RectangleRounded(
                width=width,
                height=depth,
                radius=BIN_CORNER_RADIUS
            )
            RectangleRounded(
                width=width - lip_outer * 2,
                height=depth - lip_outer * 2,
                radius=BIN_CORNER_RADIUS - lip_outer,
                mode=Mode.SUBTRACT
            )
        extrude(amount=-lip_height, mode=Mode.SUBTRACT)
    
    def _add_base_profile(self, builder, width: float, depth: float):
        """Adiciona perfil de encaixe na base para conectar ao baseplate."""
        # Chanfro na base para encaixe suave
        base_inset = 0.8
        base_height = 4.65
        
        with BuildSketch(Plane.XY):
            RectangleRounded(
                width=width,
                height=depth,
                radius=BIN_CORNER_RADIUS
            )
            RectangleRounded(
                width=width - base_inset * 2,
                height=depth - base_inset * 2,
                radius=BIN_CORNER_RADIUS - base_inset,
                mode=Mode.SUBTRACT
            )
        extrude(amount=base_height, mode=Mode.SUBTRACT)
    
    def _add_scoop(
        self,
        builder,
        internal_width: float,
        internal_depth: float,
        internal_height: float
    ):
        """Adiciona rampa (scoop) para facilitar pegar itens."""
        scoop_depth = internal_depth * 0.3
        
        with BuildSketch(Plane.XZ.offset(-internal_depth / 2 + BIN_WALL_THICKNESS)):
            with Locations([(0, BIN_BOTTOM_THICKNESS + internal_height / 4)]):
                Ellipse(
                    x_radius=internal_width / 3,
                    y_radius=internal_height / 3
                )
        extrude(amount=scoop_depth, mode=Mode.SUBTRACT)
    
    def _add_label_area(self, builder, width: float, height: float):
        """Adiciona área inclinada para etiqueta na frente."""
        label_width = width * 0.8
        label_height = 12  # mm
        label_depth = 0.6  # profundidade do rebaixo
        
        # Área rebaixada para etiqueta
        with BuildSketch(Plane.XZ.offset(-width / 2)):
            with Locations([(0, height - label_height / 2 - 2)]):
                RectangleRounded(
                    width=label_width,
                    height=label_height,
                    radius=1.5
                )
        extrude(amount=label_depth, mode=Mode.SUBTRACT)
    
    def generate_divider_bin(
        self,
        units_x: int,
        units_y: int,
        height_units: int = 3,
        divisions_x: int = 1,
        divisions_y: int = 1
    ) -> Part:
        """
        Gera um bin com divisórias internas.
        
        Args:
            units_x: Unidades no eixo X
            units_y: Unidades no eixo Y
            height_units: Altura em unidades
            divisions_x: Número de divisões no eixo X
            divisions_y: Número de divisões no eixo Y
            
        Returns:
            Objeto Part representando o bin com divisórias
        """
        # Primeiro, gera o bin base
        base_bin = self.generate_bin(units_x, units_y, height_units)
        
        width = units_x * GRID_UNIT - 0.5
        depth = units_y * GRID_UNIT - 0.5
        height = height_units * GRID_HEIGHT_UNIT
        
        internal_width = width - BIN_WALL_THICKNESS * 2
        internal_depth = depth - BIN_WALL_THICKNESS * 2
        internal_height = height - BIN_BOTTOM_THICKNESS
        
        divider_thickness = 1.0  # mm
        
        with BuildPart() as bin_with_dividers:
            add(base_bin)
            
            # Divisórias no eixo X
            if divisions_x > 1:
                cell_width = internal_width / divisions_x
                for i in range(1, divisions_x):
                    x_pos = -internal_width / 2 + cell_width * i
                    with BuildSketch(Plane.YZ.offset(x_pos)):
                        Rectangle(internal_depth, internal_height)
                    extrude(amount=divider_thickness, both=True)
            
            # Divisórias no eixo Y
            if divisions_y > 1:
                cell_depth = internal_depth / divisions_y
                for i in range(1, divisions_y):
                    y_pos = -internal_depth / 2 + cell_depth * i
                    with BuildSketch(Plane.XZ.offset(y_pos)):
                        Rectangle(internal_width, internal_height)
                    extrude(amount=divider_thickness, both=True)
        
        return bin_with_dividers.part

"""
Utilitários para exportação de arquivos STL e ZIP
"""

import io
import zipfile
import tempfile
import os
from typing import List, Tuple, Optional
from build123d import Compound, Part, export_stl


def export_to_stl_bytes(shape: Compound | Part, tolerance: float = 0.001) -> bytes:
    """
    Exporta um shape para bytes STL.
    
    Args:
        shape: Objeto Compound ou Part do build123d
        tolerance: Tolerância para tessellation
        
    Returns:
        Bytes do arquivo STL
    """
    with tempfile.NamedTemporaryFile(suffix='.stl', delete=False) as tmp:
        tmp_path = tmp.name
    
    try:
        export_stl(shape, tmp_path, tolerance=tolerance)
        with open(tmp_path, 'rb') as f:
            return f.read()
    finally:
        if os.path.exists(tmp_path):
            os.remove(tmp_path)


def create_download_zip(
    baseplate: Optional[Compound | Part] = None,
    bins: Optional[List[Tuple[str, Part]]] = None,
    baseplate_filename: str = "baseplate_full_set_a1mini.stl"
) -> bytes:
    """
    Cria arquivo ZIP contendo baseplate e bins.
    
    Args:
        baseplate: Compound com todas as peças do baseplate
        bins: Lista de tuplas (nome_arquivo, Part) para os bins
        baseplate_filename: Nome do arquivo do baseplate no ZIP
        
    Returns:
        Bytes do arquivo ZIP
    """
    zip_buffer = io.BytesIO()
    
    with zipfile.ZipFile(zip_buffer, 'w', zipfile.ZIP_DEFLATED) as zf:
        # Adiciona baseplate
        if baseplate is not None:
            stl_bytes = export_to_stl_bytes(baseplate)
            zf.writestr(baseplate_filename, stl_bytes)
        
        # Adiciona bins
        if bins:
            for filename, bin_part in bins:
                stl_bytes = export_to_stl_bytes(bin_part)
                zf.writestr(filename, stl_bytes)
    
    zip_buffer.seek(0)
    return zip_buffer.getvalue()


def get_file_size_str(size_bytes: int) -> str:
    """Converte bytes para string legível."""
    if size_bytes < 1024:
        return f"{size_bytes} B"
    elif size_bytes < 1024 * 1024:
        return f"{size_bytes / 1024:.1f} KB"
    else:
        return f"{size_bytes / (1024 * 1024):.1f} MB"

import pytest
from gridfinity.bins import BinGenerator
from gridfinity.constants import GRID_UNIT, A1_MINI_MAX_UNITS


class TestBinGenerator:
    """Testes para BinGenerator."""
    
    def test_validate_size_valid(self):
        """Testa validação de tamanho válido."""
        gen = BinGenerator(optimize_for_a1_mini=True)
        
        is_valid, msg = gen.validate_size(2, 2)
        assert is_valid
        assert "✅" in msg or "compatível" in msg.lower()
    
    def test_validate_size_max_limit(self):
        """Testa validação no limite máximo."""
        gen = BinGenerator(optimize_for_a1_mini=True)
        
        is_valid, msg = gen.validate_size(4, 4)
        assert is_valid
    
    def test_validate_size_invalid(self):
        """Testa validação de tamanho inválido."""
        gen = BinGenerator(optimize_for_a1_mini=True)
        
        is_valid, msg = gen.validate_size(5, 4)
        assert not is_valid
        assert "alerta" in msg.lower() or "⚠️" in msg
    
    def test_validate_size_both_exceed(self):
        """Testa validação quando ambos eixos excedem."""
        gen = BinGenerator(optimize_for_a1_mini=True)
        
        is_valid, msg = gen.validate_size(5, 5)
        assert not is_valid
    
    def test_validate_size_no_optimization(self):
        """Testa validação sem otimização."""
        gen = BinGenerator(optimize_for_a1_mini=False)
        
        is_valid, msg = gen.validate_size(10, 10)
        assert is_valid
    
    def test_bin_dimensions(self):
        """Testa cálculo de dimensões do bin."""
        # Dimensões esperadas para bin 2x2x3
        expected_width = 2 * GRID_UNIT - 0.5  # 83.5mm
        expected_depth = 2 * GRID_UNIT - 0.5  # 83.5mm
        
        assert expected_width == 83.5
        assert expected_depth == 83.5

"""
Testes unitários para o módulo Gridfinity Generator
"""

import pytest
from gridfinity.constants import (
    GRID_UNIT, A1_MINI_MAX_UNITS, GRID_HEIGHT_UNIT
)


class TestConstants:
    """Testes para as constantes do Gridfinity."""
    
    def test_grid_unit_value(self):
        """Verifica se a unidade de grid está correta."""
        assert GRID_UNIT == 42.0
    
    def test_a1_mini_max_units(self):
        """Verifica limite da A1 Mini."""
        assert A1_MINI_MAX_UNITS == 4
        # 4 unidades * 42mm = 168mm < 180mm (volume A1 Mini)
        assert A1_MINI_MAX_UNITS * GRID_UNIT < 180
    
    def test_grid_height_unit(self):
        """Verifica altura da unidade."""
        assert GRID_HEIGHT_UNIT == 7.0
